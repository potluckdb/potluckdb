import { Aside, Card, CardGrid } from '@astrojs/starlight/components';

<Aside type="caution" title="Pre-release: no sync yet">
The 0.8 series is built in the open. The current pre-releases are the SQLite
drop-in library and its language packages; nothing syncs yet. Sync comes in
later releases: see [what works today](/start/status/).
Not for production use before 1.0.
</Aside>

<CardGrid>
  <Card title="Drop-in" icon="puzzle">
    PotluckDB replaces the standard SQLite library. Your code, drivers and
    ORMs keep working.
  </Card>
  <Card title="Local and fast" icon="rocket">
    Reads and writes stay on the device. Sync, coming in later releases, will
    work in the background.
  </Card>
  <Card title="Your bucket" icon="seti:db">
    Sync will go through S3, Cloudflare R2, Google Cloud Storage, Azure Blob
    or a folder. There is no PotluckDB server.
  </Card>
</CardGrid>
